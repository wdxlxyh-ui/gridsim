# 数据持久化（1 小时滚动窗口）实施计划 v2

> **给实施者：** 本计划按任务顺序执行，每个步骤用 `- [ ]` 勾选跟踪。严格遵循 TDD：先写失败测试 → 运行确认失败 → 最小实现 → 运行确认通过 → 提交。跳过「确认失败」这一步等于没做 TDD。

**Goal:** 让 GridSim 的测点数据落入本地 SQLite，界面打开时显示数据库中最近 1 小时的真实数据而非空白；实例重启后自动变化策略不再丢失。

**Architecture:** 新增 `internal/store` 包承担全部数据访问。采样经异步批量写入管道落库，时序数据按实例独立表存储（`ts_<instance_id>`），固定 1 小时滚动窗口，超期数据在每分钟清理时丢弃，数据库体积稳定在约 30 MB。跨实例查询通过按需拼接 `UNION ALL` 完成。数据库仅作数据层，`instances.json` 与点表 `.xlsx` 仍是配置权威源，因此数据库损坏或关闭都不会丢配置。

**Tech Stack:** Go 1.25 + `modernc.org/sqlite` v1.56.0（纯 Go，无 CGO）+ Vue 3 + TypeScript + Vitest（新增）+ ECharts

**本计划范围：** 后端 store 包、写入管道、策略持久化与启动恢复、历史查询与快照 API、前端能力探测与首屏快照、趋势历史回填。

**不在本计划范围（下一版计划）：** 趋势模板与面板配置迁移到服务端、事件与控制审计表、独立历史查询页。这些依赖本计划的基础设施，单独成计划交付。

## 前置：为什么测试要在远程跑

本地 Windows 环境不具备编译能力，代码在本地编写，编译与测试在编译服务器执行。这不是可选项，是本计划所有测试步骤的执行前提。

已实测确认的两个环境事实：

1. 编译服务器 `/tmp` 挂载带 `noexec`，`go test` 会因无法执行测试二进制而全部失败，报 `fork/exec ... permission denied`。**必须设置 `TMPDIR`**
2. 测试在独立克隆 `/root/IEC-SIM/gridsim-test` 中运行，与构建目录 `/root/IEC-SIM/iec104-sim-master` 隔离，避免测试产物污染构建与部署流程

- [ ] **Step 5: 写最小实现**

创建 `internal/store/db.go`：

```go
// Package store 提供 GridSim 的本地数据持久化能力。
// 时序数据采用固定滚动窗口，超出保留期的数据在清理时丢弃。
package store

import (
	"database/sql"
	"fmt"
	"regexp"
	"time"

	_ "modernc.org/sqlite"
)

// reInstanceID 限制实例 ID 字符集，防止表名拼接被注入。
// 实例 ID 由 manager.generateID() 生成为小写十六进制。
var reInstanceID = regexp.MustCompile(`^[0-9a-f]{6,32}$`)

// tsTableName 返回实例对应的时序表名，非法 ID 一律拒绝。
// 所有拼接表名的位置都必须经过本函数。
func tsTableName(instanceID string) (string, error) {
	if !reInstanceID.MatchString(instanceID) {
		return "", fmt.Errorf("store: invalid instance id %q", instanceID)
	}
	return "ts_" + instanceID, nil
}

// Sample 是一条时序采样。
type Sample struct {
	IOA       uint32
	TS        int64 // Unix 毫秒
	Value     float64
	BoolValue bool
	IntValue  int32
	QDS       uint8
}

// DB 包装底层连接。
type DB struct {
	sql *sql.DB
}

// SQL 暴露底层连接，供查询与测试使用。
func (d *DB) SQL() *sql.DB { return d.sql }

// Close 关闭前做一次 WAL 检查点，保证文件自洽、备份可用。
func (d *DB) Close() error {
	if d == nil || d.sql == nil {
		return nil
	}
	_, _ = d.sql.Exec(`PRAGMA wal_checkpoint(TRUNCATE)`)
	return d.sql.Close()
}

// dsnParams 说明：
//   journal_mode=WAL   读写并发，轮询查询不阻塞采样写入
//   synchronous=NORMAL 兼顾安全与吞吐
//   busy_timeout=5000  锁等待 5s，避免瞬时并发直接报错
//   foreign_keys=ON    启用外键约束
const dsnParams = "?_pragma=journal_mode(WAL)" +
	"&_pragma=synchronous(NORMAL)" +
	"&_pragma=busy_timeout(5000)" +
	"&_pragma=foreign_keys(ON)"

// Open 打开或创建数据库，并执行 schema 迁移。
func Open(path string) (*DB, error) {
	sdb, err := sql.Open("sqlite", "file:"+path+dsnParams)
	if err != nil {
		return nil, fmt.Errorf("store: open %s: %w", path, err)
	}
	// SQLite 为单写者模型，限制为 1 条连接可避免写锁竞争与 SQLITE_BUSY。
	sdb.SetMaxOpenConns(1)
	if err := sdb.Ping(); err != nil {
		sdb.Close()
		return nil, fmt.Errorf("store: ping %s: %w", path, err)
	}
	d := &DB{sql: sdb}
	if err := d.migrate(); err != nil {
		sdb.Close()
		return nil, err
	}
	return d, nil
}

// migrate 顺序执行迁移，已应用的版本跳过。每个迁移在单事务内完成。
func (d *DB) migrate() error {
	if _, err := d.sql.Exec(`CREATE TABLE IF NOT EXISTS schema_version (
		version     INTEGER PRIMARY KEY,
		applied_at  INTEGER NOT NULL,
		description TEXT
	)`); err != nil {
		return fmt.Errorf("store: create schema_version: %w", err)
	}

	var cur int
	if err := d.sql.QueryRow(`SELECT COALESCE(MAX(version),0) FROM schema_version`).Scan(&cur); err != nil {
		return fmt.Errorf("store: read schema_version: %w", err)
	}

	for _, m := range migrations {
		if m.Version <= cur {
			continue
		}
		tx, err := d.sql.Begin()
		if err != nil {
			return fmt.Errorf("store: begin migration %d: %w", m.Version, err)
		}
		for _, stmt := range m.Statements {
			if _, err := tx.Exec(stmt); err != nil {
				tx.Rollback()
				return fmt.Errorf("store: migration %d failed: %w", m.Version, err)
			}
		}
		if _, err := tx.Exec(
			`INSERT INTO schema_version(version,applied_at,description) VALUES(?,?,?)`,
			m.Version, time.Now().UnixMilli(), m.Description,
		); err != nil {
			tx.Rollback()
			return fmt.Errorf("store: record migration %d: %w", m.Version, err)
		}
		if err := tx.Commit(); err != nil {
			return fmt.Errorf("store: commit migration %d: %w", m.Version, err)
		}
	}
	return nil
}

// CreateInstanceTable 为实例建时序表，幂等。
func (d *DB) CreateInstanceTable(instanceID string) error {
	tbl, err := tsTableName(instanceID)
	if err != nil {
		return err
	}
	stmts := []string{
		fmt.Sprintf(`CREATE TABLE IF NOT EXISTS %s (
			ioa        INTEGER NOT NULL,
			ts         INTEGER NOT NULL,
			value      REAL,
			bool_value INTEGER,
			int_value  INTEGER,
			qds        INTEGER DEFAULT 0,
			PRIMARY KEY (ioa, ts)
		) WITHOUT ROWID`, tbl),
		fmt.Sprintf(`CREATE INDEX IF NOT EXISTS idx_%s_ts ON %s(ts)`, tbl, tbl),
	}
	for _, s := range stmts {
		if _, err := d.sql.Exec(s); err != nil {
			return fmt.Errorf("store: create %s: %w", tbl, err)
		}
	}
	return nil
}

// InsertSamples 单事务批量写入。空切片直接返回。
func (d *DB) InsertSamples(instanceID string, samples []Sample) error {
	if len(samples) == 0 {
		return nil
	}
	tbl, err := tsTableName(instanceID)
	if err != nil {
		return err
	}
	tx, err := d.sql.Begin()
	if err != nil {
		return fmt.Errorf("store: begin insert: %w", err)
	}
	st, err := tx.Prepare(fmt.Sprintf(
		`INSERT OR REPLACE INTO %s(ioa,ts,value,bool_value,int_value,qds) VALUES(?,?,?,?,?,?)`, tbl))
	if err != nil {
		tx.Rollback()
		return fmt.Errorf("store: prepare insert: %w", err)
	}
	defer st.Close()
	for _, s := range samples {
		b := 0
		if s.BoolValue {
			b = 1
		}
		if _, err := st.Exec(s.IOA, s.TS, s.Value, b, s.IntValue, s.QDS); err != nil {
			tx.Rollback()
			return fmt.Errorf("store: insert ioa=%d: %w", s.IOA, err)
		}
	}
	if err := tx.Commit(); err != nil {
		return fmt.Errorf("store: commit insert: %w", err)
	}
	return nil
}
```

```go
// Cleanup 丢弃超出滚动窗口的样本，返回删除行数。
// retentionMinutes 必须为正数：传 0 会清空全表，属于危险配置，直接拒绝。
func (d *DB) Cleanup(instanceID string, retentionMinutes int) (int64, error) {
	tbl, err := tsTableName(instanceID)
	if err != nil {
		return 0, err
	}
	if retentionMinutes <= 0 {
		return 0, fmt.Errorf("store: retentionMinutes must be > 0, got %d", retentionMinutes)
	}
	cutoff := time.Now().UnixMilli() - int64(retentionMinutes)*60_000
	res, err := d.sql.Exec(fmt.Sprintf(`DELETE FROM %s WHERE ts < ?`, tbl), cutoff)
	if err != nil {
		return 0, fmt.Errorf("store: cleanup %s: %w", tbl, err)
	}
	n, _ := res.RowsAffected()
	// 删除会产生大量 WAL 页，做一次被动检查点避免 WAL 无界增长。
	// 不做 VACUUM：空闲页由后续写入复用，稳态文件大小自然稳定。
	if n > 0 {
		_, _ = d.sql.Exec(`PRAGMA wal_checkpoint(PASSIVE)`)
	}
	return n, nil
}
```

```go
package store

import (
	"fmt"
	"log"
	"sync"
	"sync/atomic"
	"time"
)

// queued 是入队的采样，附带实例 ID。
type queued struct {
	instanceID string
	sample     Sample
}

// WriterOptions 控制写入与清理节奏。零值字段由 NewWriter 填默认值。
type WriterOptions struct {
	QueueSize        int           // 队列容量，默认 8192
	FlushInterval    time.Duration // 刷盘间隔，默认 1s
	MaxBatch         int           // 批量上限，默认 2000
	CleanupInterval  time.Duration // 清理间隔，默认 60s
	RetentionMinutes int           // 滚动窗口，默认 60
	Deadband         bool          // 仅记录变化值，默认关闭
}

func (o *WriterOptions) applyDefaults() {
	if o.QueueSize <= 0 {
		o.QueueSize = 8192
	}
	if o.FlushInterval <= 0 {
		o.FlushInterval = time.Second
	}
	if o.MaxBatch <= 0 {
		o.MaxBatch = 2000
	}
	if o.CleanupInterval <= 0 {
		o.CleanupInterval = 60 * time.Second
	}
	if o.RetentionMinutes <= 0 {
		o.RetentionMinutes = 60
	}
}

// Writer 把采样异步批量写入数据库。
// 关键约束：Push 永不阻塞调用方，队列满时丢弃并计数，
// 保证协议栈与仿真引擎不会因数据库变慢而受影响。
type Writer struct {
	db   *DB
	opt  WriterOptions
	ch   chan queued
	done chan struct{}
	wg   sync.WaitGroup

	dropped atomic.Int64
	written atomic.Int64

	startOnce sync.Once
	stopOnce  sync.Once

	// lastVal 是死区比较基准，仅写入 goroutine 访问，无需加锁。
	lastVal map[string]float64
}

// NewWriter 创建写入器，不自动启动。
func NewWriter(db *DB, opt WriterOptions) *Writer {
	opt.applyDefaults()
	return &Writer{
		db:      db,
		opt:     opt,
		ch:      make(chan queued, opt.QueueSize),
		done:    make(chan struct{}),
		lastVal: make(map[string]float64),
	}
}

// Start 启动后台写入 goroutine，重复调用无副作用。
func (w *Writer) Start() {
	w.startOnce.Do(func() {
		w.wg.Add(1)
		go w.loop()
	})
}

// Stop 刷盘并停止后台 goroutine，重复调用无副作用。
func (w *Writer) Stop() {
	w.stopOnce.Do(func() {
		close(w.done)
		w.wg.Wait()
	})
}

// Push 非阻塞入队。队列满时丢弃当前样本并计数。
func (w *Writer) Push(instanceID string, s Sample) {
	if w == nil {
		return
	}
	select {
	case w.ch <- queued{instanceID: instanceID, sample: s}:
	default:
		w.dropped.Add(1)
	}
}

// Dropped 返回累计丢弃数，供 db/status 暴露。
func (w *Writer) Dropped() int64 { return w.dropped.Load() }

// Written 返回累计写入行数。
func (w *Writer) Written() int64 { return w.written.Load() }

func (w *Writer) loop() {
	defer w.wg.Done()

	flushTick := time.NewTicker(w.opt.FlushInterval)
	defer flushTick.Stop()
	cleanTick := time.NewTicker(w.opt.CleanupInterval)
	defer cleanTick.Stop()

	// batch 按实例分组，刷盘时每个实例一个事务。
	batch := make(map[string][]Sample)
	n := 0

	flush := func() {
		if n == 0 {
			return
		}
		for inst, samples := range batch {
			if err := w.db.InsertSamples(inst, samples); err != nil {
				// 写入失败不向上传播，仅记录，保证仿真不受影响。
				log.Printf("[store] flush instance=%s rows=%d failed: %v", inst, len(samples), err)
				continue
			}
			w.written.Add(int64(len(samples)))
		}
		batch = make(map[string][]Sample)
		n = 0
	}

	for {
		select {
		case q := <-w.ch:
			if w.opt.Deadband && w.skipByDeadband(q) {
				continue
			}
			batch[q.instanceID] = append(batch[q.instanceID], q.sample)
			n++
			if n >= w.opt.MaxBatch {
				flush()
			}

		case <-flushTick.C:
			flush()

		case <-cleanTick.C:
			// 先刷盘，避免清理误判刚入队的数据；
			// 清理与写入在同一 goroutine，天然串行，不会争抢 WAL 写锁。
			flush()
			w.cleanupAll()

		case <-w.done:
			// 退出前把队列里剩余样本排空并落库。
			for {
				select {
				case q := <-w.ch:
					if w.opt.Deadband && w.skipByDeadband(q) {
						continue
					}
					batch[q.instanceID] = append(batch[q.instanceID], q.sample)
					n++
				default:
					flush()
					return
				}
			}
		}
	}
}

// skipByDeadband 判断是否因值未变化而跳过。品质位变化一律不跳过。
func (w *Writer) skipByDeadband(q queued) bool {
	key := q.instanceID + ":" + fmt.Sprint(q.sample.IOA)
	prev, ok := w.lastVal[key]
	if ok && prev == q.sample.Value && q.sample.QDS == 0 {
		return true
	}
	w.lastVal[key] = q.sample.Value
	return false
}

// cleanupAll 对已知实例逐表清理。实例集合取自 lastVal 与数据库中的表。
func (w *Writer) cleanupAll() {
	insts, err := w.db.ListInstanceTables()
	if err != nil {
		log.Printf("[store] list instance tables failed: %v", err)
		return
	}
	for _, inst := range insts {
		if _, err := w.db.Cleanup(inst, w.opt.RetentionMinutes); err != nil {
			log.Printf("[store] cleanup instance=%s failed: %v", inst, err)
		}
	}
}
```

```go
package store

import (
	"fmt"
	"strconv"
	"strings"
)

// Series 是单个测点在一段时间内的采样序列。
// Samples 用 [时间戳, 值] 二元数组，直接对应 ECharts time 轴格式，
// 相比对象数组可减少约 60% 传输体积。
type Series struct {
	IOA     uint32      `json:"ioa"`
	Samples [][2]float64 `json:"samples"`
}

// LatestSample 是某测点的最新一条采样。
type LatestSample struct {
	IOA       uint32  `json:"ioa"`
	TS        int64   `json:"ts"`
	Value     float64 `json:"value"`
	BoolValue bool    `json:"bool_value"`
	IntValue  int32   `json:"int_value"`
	QDS       uint8   `json:"qds"`
}

// ioaPlaceholders 生成 IN 子句占位符与参数，避免拼接数值。
func ioaPlaceholders(ioas []uint32) (string, []any) {
	parts := make([]string, len(ioas))
	args := make([]any, len(ioas))
	for i, v := range ioas {
		parts[i] = "?"
		args[i] = int64(v)
	}
	return strings.Join(parts, ","), args
}

// QuerySeries 查询指定测点在 [from, to] 内的采样。
// limit > 0 时保留最新的 limit 条，返回结果仍按时间升序。
func (d *DB) QuerySeries(instanceID string, ioas []uint32, from, to int64, limit int) ([]Series, error) {
	tbl, err := tsTableName(instanceID)
	if err != nil {
		return nil, err
	}
	if len(ioas) == 0 {
		return nil, nil
	}

	byIOA := make(map[uint32][][2]float64, len(ioas))
	for _, ioa := range ioas {
		// 每个 IOA 单独查询，便于按 IOA 施加 limit 并利用 (ioa,ts) 聚簇主键。
		q := fmt.Sprintf(
			`SELECT ts, value, bool_value, int_value FROM %s
			 WHERE ioa = ? AND ts BETWEEN ? AND ?
			 ORDER BY ts DESC`, tbl)
		args := []any{int64(ioa), from, to}
		if limit > 0 {
			q += " LIMIT " + strconv.Itoa(limit)
		}

		rows, err := d.sql.Query(q, args...)
		if err != nil {
			return nil, fmt.Errorf("store: query series ioa=%d: %w", ioa, err)
		}

		var desc [][2]float64
		for rows.Next() {
			var ts int64
			var val, bval, ival any
			if err := rows.Scan(&ts, &val, &bval, &ival); err != nil {
				rows.Close()
				return nil, fmt.Errorf("store: scan series ioa=%d: %w", ioa, err)
			}
			desc = append(desc, [2]float64{float64(ts), coalesceNumeric(val, bval, ival)})
		}
		if err := rows.Err(); err != nil {
			rows.Close()
			return nil, fmt.Errorf("store: iterate series ioa=%d: %w", ioa, err)
		}
		rows.Close()

		// 由 DESC 反转为升序
		for i, j := 0, len(desc)-1; i < j; i, j = i+1, j-1 {
			desc[i], desc[j] = desc[j], desc[i]
		}
		if len(desc) > 0 {
			byIOA[ioa] = desc
		}
	}

	out := make([]Series, 0, len(byIOA))
	for _, ioa := range ioas {
		if s, ok := byIOA[ioa]; ok {
			out = append(out, Series{IOA: ioa, Samples: s})
		}
	}
	return out, nil
}

// QueryLatest 返回实例内每个测点的最新一条采样。
func (d *DB) QueryLatest(instanceID string) (map[uint32]LatestSample, error) {
	tbl, err := tsTableName(instanceID)
	if err != nil {
		return nil, err
	}
	// 按 (ioa, ts) 主键聚簇，MAX(ts) 分组可走索引。
	q := fmt.Sprintf(
		`SELECT t.ioa, t.ts, t.value, t.bool_value, t.int_value, t.qds
		 FROM %s t
		 JOIN (SELECT ioa, MAX(ts) AS mts FROM %s GROUP BY ioa) m
		   ON m.ioa = t.ioa AND m.mts = t.ts`, tbl, tbl)

	rows, err := d.sql.Query(q)
	if err != nil {
		return nil, fmt.Errorf("store: query latest: %w", err)
	}
	defer rows.Close()

	out := make(map[uint32]LatestSample)
	for rows.Next() {
		var ioa int64
		var ts int64
		var qds int64
		var val, bval, ival any
		if err := rows.Scan(&ioa, &ts, &val, &bval, &ival, &qds); err != nil {
			return nil, fmt.Errorf("store: scan latest: %w", err)
		}
		ls := LatestSample{
			IOA: uint32(ioa),
			TS:  ts,
			QDS: uint8(qds),
		}
		ls.Value = coalesceNumeric(val, bval, ival)
		if b, ok := toInt64(bval); ok {
			ls.BoolValue = b != 0
		}
		if iv, ok := toInt64(ival); ok {
			ls.IntValue = int32(iv)
		}
		out[uint32(ioa)] = ls
	}
	return out, rows.Err()
}

// coalesceNumeric 按 value → int_value → bool_value 优先级取出可用数值。
// 不同点类型只有一个字段有值，其余为 NULL。
func coalesceNumeric(val, bval, ival any) float64 {
	if f, ok := toFloat64(val); ok {
		return f
	}
	if i, ok := toInt64(ival); ok {
		return float64(i)
	}
	if b, ok := toInt64(bval); ok {
		return float64(b)
	}
	return 0
}

func toFloat64(v any) (float64, bool) {
	switch t := v.(type) {
	case float64:
		return t, true
	case int64:
		return float64(t), true
	default:
		return 0, false
	}
}

func toInt64(v any) (int64, bool) {
	switch t := v.(type) {
	case int64:
		return t, true
	case float64:
		return int64(t), true
	default:
		return 0, false
	}
}
```

```go
// CrossRef 指向某实例的某个测点。
type CrossRef struct {
	InstanceID string `json:"instance_id"`
	IOA        uint32 `json:"ioa"`
}

// CrossSeries 是带实例标识的采样序列。
type CrossSeries struct {
	InstanceID string       `json:"instance_id"`
	IOA        uint32       `json:"ioa"`
	Samples    [][2]float64 `json:"samples"`
}

// QueryCross 跨实例查询多个测点。
// 表不存在（实例已删除）时跳过该测点，不让整个查询失败。
// 非法实例 ID 直接拒绝，不做跳过处理，避免掩盖调用方错误。
func (d *DB) QueryCross(refs []CrossRef, from, to int64, limit int) ([]CrossSeries, error) {
	// 先整体校验 ID，任何非法立即返回错误。
	byInst := make(map[string][]uint32)
	order := make([]CrossRef, 0, len(refs))
	for _, r := range refs {
		if _, err := tsTableName(r.InstanceID); err != nil {
			return nil, err
		}
		byInst[r.InstanceID] = append(byInst[r.InstanceID], r.IOA)
		order = append(order, r)
	}

	existing, err := d.ListInstanceTables()
	if err != nil {
		return nil, err
	}
	have := make(map[string]bool, len(existing))
	for _, id := range existing {
		have[id] = true
	}

	// 按实例批量查询，结果放入索引便于按请求顺序输出。
	idx := make(map[string][][2]float64)
	for inst, ioas := range byInst {
		if !have[inst] {
			continue // 实例表不存在，跳过
		}
		series, err := d.QuerySeries(inst, ioas, from, to, limit)
		if err != nil {
			return nil, err
		}
		for _, s := range series {
			idx[inst+":"+strconv.FormatUint(uint64(s.IOA), 10)] = s.Samples
		}
	}

	out := make([]CrossSeries, 0, len(order))
	seen := make(map[string]bool, len(order))
	for _, r := range order {
		key := r.InstanceID + ":" + strconv.FormatUint(uint64(r.IOA), 10)
		if seen[key] {
			continue
		}
		seen[key] = true
		if s, ok := idx[key]; ok {
			out = append(out, CrossSeries{InstanceID: r.InstanceID, IOA: r.IOA, Samples: s})
		}
	}
	return out, nil
}
```

```go
package store

import (
	"fmt"
	"time"
)

// StrategyRecord 是一条自动变化策略的持久化形态。
// ParamsJSON 保留原始 JSON 而不拆列：StrategyParams 有 30 余个可选字段
// 且随策略类型扩展，拆列会带来大量空列与频繁 schema 变更。
type StrategyRecord struct {
	IOA        uint32 `json:"ioa"`
	Strategy   string `json:"strategy"`
	Enabled    bool   `json:"enabled"`
	ParamsJSON string `json:"params_json"`
	UpdatedAt  int64  `json:"updated_at"`
}

// SaveStrategies 覆盖写入指定实例的策略。同 IOA 覆盖，单事务提交。
func (d *DB) SaveStrategies(instanceID string, recs []StrategyRecord) error {
	// 借用表名校验做 ID 合法性检查，策略表本身不按实例分表。
	if _, err := tsTableName(instanceID); err != nil {
		return err
	}
	if len(recs) == 0 {
		return nil
	}

	tx, err := d.sql.Begin()
	if err != nil {
		return fmt.Errorf("store: begin save strategies: %w", err)
	}
	st, err := tx.Prepare(`INSERT INTO auto_change_configs
		(instance_id, ioa, strategy, enabled, params_json, updated_at)
		VALUES(?,?,?,?,?,?)
		ON CONFLICT(instance_id, ioa) DO UPDATE SET
			strategy    = excluded.strategy,
			enabled     = excluded.enabled,
			params_json = excluded.params_json,
			updated_at  = excluded.updated_at`)
	if err != nil {
		tx.Rollback()
		return fmt.Errorf("store: prepare save strategies: %w", err)
	}
	defer st.Close()

	now := time.Now().UnixMilli()
	for _, r := range recs {
		en := 0
		if r.Enabled {
			en = 1
		}
		ts := r.UpdatedAt
		if ts == 0 {
			ts = now
		}
		if _, err := st.Exec(instanceID, int64(r.IOA), r.Strategy, en, r.ParamsJSON, ts); err != nil {
			tx.Rollback()
			return fmt.Errorf("store: save strategy ioa=%d: %w", r.IOA, err)
		}
	}
	if err := tx.Commit(); err != nil {
		return fmt.Errorf("store: commit save strategies: %w", err)
	}
	return nil
}

// LoadStrategies 读取指定实例的全部策略，按 IOA 升序。
func (d *DB) LoadStrategies(instanceID string) ([]StrategyRecord, error) {
	if _, err := tsTableName(instanceID); err != nil {
		return nil, err
	}
	rows, err := d.sql.Query(
		`SELECT ioa, strategy, enabled, params_json, updated_at
		 FROM auto_change_configs WHERE instance_id = ? ORDER BY ioa`, instanceID)
	if err != nil {
		return nil, fmt.Errorf("store: load strategies: %w", err)
	}
	defer rows.Close()

	var out []StrategyRecord
	for rows.Next() {
		var ioa, en, ts int64
		var strategy, params string
		if err := rows.Scan(&ioa, &strategy, &en, &params, &ts); err != nil {
			return nil, fmt.Errorf("store: scan strategy: %w", err)
		}
		out = append(out, StrategyRecord{
			IOA:        uint32(ioa),
			Strategy:   strategy,
			Enabled:    en != 0,
			ParamsJSON: params,
			UpdatedAt:  ts,
		})
	}
	return out, rows.Err()
}

// DeleteStrategy 删除单个测点的策略。
func (d *DB) DeleteStrategy(instanceID string, ioa uint32) error {
	if _, err := tsTableName(instanceID); err != nil {
		return err
	}
	if _, err := d.sql.Exec(
		`DELETE FROM auto_change_configs WHERE instance_id = ? AND ioa = ?`,
		instanceID, int64(ioa)); err != nil {
		return fmt.Errorf("store: delete strategy ioa=%d: %w", ioa, err)
	}
	return nil
}

// DeleteAllStrategies 删除实例的全部策略，用于实例删除时清理。
func (d *DB) DeleteAllStrategies(instanceID string) error {
	if _, err := tsTableName(instanceID); err != nil {
		return err
	}
	if _, err := d.sql.Exec(
		`DELETE FROM auto_change_configs WHERE instance_id = ?`, instanceID); err != nil {
		return fmt.Errorf("store: delete all strategies: %w", err)
	}
	return nil
}
```

```go
// Store 是数据持久化的统一入口。
// 关闭数据库时注入 Noop，调用方无需判空或判开关。
type Store interface {
	// Enabled 报告数据库是否可用，供 API 暴露能力给前端。
	Enabled() bool

	CreateInstanceTable(instanceID string) error

	DropInstanceData(instanceID string) error

	LoadStrategies(instanceID string) ([]StrategyRecord, error)

	Close() error
```

```go
// Impl 是基于 SQLite 的真实实现。
type Impl struct {
	db     *DB
	writer *Writer
	path   string
	opt    WriterOptions
}

// Config 是打开数据库所需的参数。
type Config struct {
	Path             string
	RetentionMinutes int
	SampleIntervalMs int
	CleanupIntervalS int
	Deadband         bool
}

// New 按配置打开数据库并启动写入器。
// 打开失败时返回错误，由调用方决定是否降级为 Noop。
func New(cfg Config) (*Impl, error) {
	if cfg.RetentionMinutes <= 0 {
		cfg.RetentionMinutes = 60
	}
	if cfg.SampleIntervalMs <= 0 {
		cfg.SampleIntervalMs = 1000
	}
	if cfg.CleanupIntervalS <= 0 {
		cfg.CleanupIntervalS = 60
	}

	db, err := Open(cfg.Path)
	if err != nil {
		return nil, err
	}

	opt := WriterOptions{
		FlushInterval:    time.Duration(cfg.SampleIntervalMs) * time.Millisecond,
		CleanupInterval:  time.Duration(cfg.CleanupIntervalS) * time.Second,
		RetentionMinutes: cfg.RetentionMinutes,
		Deadband:         cfg.Deadband,
	}
	w := NewWriter(db, opt)
	w.Start()

	log.Printf("[store] enabled path=%s retention=%dmin flush=%dms",
		cfg.Path, cfg.RetentionMinutes, cfg.SampleIntervalMs)

	return &Impl{db: db, writer: w, path: cfg.Path, opt: opt}, nil
}

func (s *Impl) Enabled() bool { return true }

func (s *Impl) CreateInstanceTable(instanceID string) error {
```

```go
	existing := eng.AllConfigs()
```

```go
// initDataStore 按模式初始化数据持久化。
// auto: 打开失败降级为空实现并告警；on: 失败直接退出；off: 直接用空实现。
func initDataStore(mode, path, configDir string, retentionMin, sampleMs int) store.Store {
	if mode == "off" {
		log.Printf("[store] disabled by --db=off")
		return store.Noop{}
	}
	if retentionMin <= 0 || retentionMin > 240 {
		log.Fatalf("[store] --db-retention-minutes 必须在 1..240 之间，当前 %d", retentionMin)
	}

	if path == "" {
		path = filepath.Join(configDir, "db", "gridsim.db")
	}
	if err := os.MkdirAll(filepath.Dir(path), 0o755); err != nil {
		if mode == "on" {
			log.Fatalf("[store] 创建目录失败: %v", err)
		}
		log.Printf("[store] 创建目录失败，降级为关闭: %v", err)
		return store.Noop{}
	}

	s, err := store.New(store.Config{
		Path:             path,
		RetentionMinutes: retentionMin,
		SampleIntervalMs: sampleMs,
	})
	if err != nil {
		if mode == "on" {
```

```go
// NewStoreHandler 创建处理器。
func NewStoreHandler(s store.Store) *StoreHandler {

// Register 注册路由。全部为新增路径，不影响既有接口。
func (h *StoreHandler) Register(mux *http.ServeMux) {

func writeJSON(w http.ResponseWriter, code int, v any) {
	w.Header().Set("Content-Type", "application/json; charset=utf-8")
	w.WriteHeader(code)
	_ = json.NewEncoder(w).Encode(v)
}

func writeErr(w http.ResponseWriter, code, errCode, msg, hint string) {
	status, _ := strconv.Atoi(code)
	if status == 0 {
		status = http.StatusBadRequest
	}
	writeJSON(w, status, map[string]any{
		"error": map[string]string{
			"code":    errCode,
			"message": msg,
			"hint":    hint,
		},
	})
}

// instanceIDFromPath 从 /api/v1/instances/{id}/... 提取实例 ID。
func instanceIDFromPath(p string) string {
	const prefix = "/api/v1/instances/"
	if !strings.HasPrefix(p, prefix) {
		return ""
	}
	rest := strings.TrimPrefix(p, prefix)
	if i := strings.IndexByte(rest, '/'); i >= 0 {
		return rest[:i]
	}
	return rest
}

// HandleDBStatus 返回数据库状态，前端据此决定是否展示历史相关功能。
func (h *StoreHandler) HandleDBStatus(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodGet {
		writeErr(w, "405", "METHOD_NOT_ALLOWED", "仅支持 GET", "")
		return
	}
	writeJSON(w, http.StatusOK, h.s.Stats())
}

// HandleLatestSnapshot 返回实例内每个测点的最新值，供详情页首屏渲染。
func (h *StoreHandler) HandleLatestSnapshot(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodGet {
		writeErr(w, "405", "METHOD_NOT_ALLOWED", "仅支持 GET", "")
		return
	}
	if !h.s.Enabled() {
		writeErr(w, "503", "DB_DISABLED", "数据持久化未启用", "使用 --db=auto 或 --db=on 启动")
		return
	}
	id := instanceIDFromPath(r.URL.Path)
	latest, err := h.s.QueryLatest(id)
	if err != nil {
		writeErr(w, "400", "QUERY_FAILED", err.Error(), "检查实例 ID 是否有效")
		return
	}

	pts := make([]store.LatestSample, 0, len(latest))
	for _, v := range latest {
		pts = append(pts, v)
	}
	sort.Slice(pts, func(i, j int) bool { return pts[i].IOA < pts[j].IOA })

	writeJSON(w, http.StatusOK, map[string]any{
		"instance_id": id,
		"points":      pts,
		"source":      "db",
	})
}

// HandleHistory 查询单实例多测点的历史序列。
// 超出保留窗口的请求会被收敛到窗口内，并通过 clamped 标记告知前端。
func (h *StoreHandler) HandleHistory(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodGet {
		writeErr(w, "405", "METHOD_NOT_ALLOWED", "仅支持 GET", "")
		return
	}
	if !h.s.Enabled() {
		writeErr(w, "503", "DB_DISABLED", "数据持久化未启用", "使用 --db=auto 或 --db=on 启动")
		return
	}

	id := instanceIDFromPath(r.URL.Path)
	q := r.URL.Query()

	ioas, err := parseIOAs(q.Get("ioas"))
	if err != nil {
		writeErr(w, "400", "INVALID_IOAS", err.Error(), "形如 ioas=16385,16386")
		return
	}

	now := time.Now().UnixMilli()
	to := parseInt64(q.Get("to"), now)
	from := parseInt64(q.Get("from"), now-15*60*1000)

	// 收敛到保留窗口：早于窗口起点的数据已被清理，查询无意义。
	retention := h.s.Stats().RetentionMinutes
	if retention <= 0 {
		retention = 60
	}
	windowStart := now - int64(retention)*60*1000
	clamped := false
	if from < windowStart {
		from = windowStart
		clamped = true
	}

	limit := int(parseInt64(q.Get("limit"), 5000))
	series, err := h.s.QuerySeries(id, ioas, from, to, limit)
	if err != nil {
		writeErr(w, "400", "QUERY_FAILED", err.Error(), "检查实例 ID 与参数")
		return
	}

	truncated := false
	for _, s := range series {
		if limit > 0 && len(s.Samples) >= limit {
			truncated = true
			break
		}
	}

	writeJSON(w, http.StatusOK, map[string]any{
		"instance_id":       id,
		"from":              from,
		"to":                to,
		"clamped":           clamped,
		"truncated":         truncated,
		"retention_minutes": retention,
		"series":            series,
	})
}

// HandleCrossHistory 跨实例查询，points=<instId>:<ioa>,<instId>:<ioa>
func (h *StoreHandler) HandleCrossHistory(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodGet {
		writeErr(w, "405", "METHOD_NOT_ALLOWED", "仅支持 GET", "")
		return
	}
	if !h.s.Enabled() {
		writeErr(w, "503", "DB_DISABLED", "数据持久化未启用", "使用 --db=auto 或 --db=on 启动")
		return
	}

	raw := r.URL.Query().Get("points")
	if raw == "" {
		writeErr(w, "400", "INVALID_POINTS", "缺少 points 参数",
			"形如 points=8e546cecbce3:16385,05438bf8b615:16386")
		return
	}

	var refs []store.CrossRef
	for _, part := range strings.Split(raw, ",") {
		kv := strings.SplitN(strings.TrimSpace(part), ":", 2)
		if len(kv) != 2 {
			writeErr(w, "400", "INVALID_POINTS", "points 格式有误: "+part,
				"形如 <instanceId>:<ioa>")
			return
		}
		ioa, err := strconv.ParseUint(kv[1], 10, 32)
		if err != nil {
			writeErr(w, "400", "INVALID_POINTS", "IOA 无效: "+kv[1], "IOA 必须为无符号整数")
			return
		}
		refs = append(refs, store.CrossRef{InstanceID: kv[0], IOA: uint32(ioa)})
	}

	now := time.Now().UnixMilli()
	to := parseInt64(r.URL.Query().Get("to"), now)
	from := parseInt64(r.URL.Query().Get("from"), now-15*60*1000)
	limit := int(parseInt64(r.URL.Query().Get("limit"), 5000))

	series, err := h.s.QueryCross(refs, from, to, limit)
	if err != nil {
		writeErr(w, "400", "QUERY_FAILED", err.Error(), "检查 points 参数中的实例 ID")
		return
	}
	writeJSON(w, http.StatusOK, map[string]any{
		"from":   from,
		"to":     to,
		"series": series,
	})
}

func parseIOAs(s string) ([]uint32, error) {
	if strings.TrimSpace(s) == "" {
		return nil, errEmptyIOAs
	}
	var out []uint32
	for _, p := range strings.Split(s, ",") {
		v, err := strconv.ParseUint(strings.TrimSpace(p), 10, 32)
		if err != nil {
			return nil, err
		}
		out = append(out, uint32(v))
	}
	return out, nil
}

func parseInt64(s string, def int64) int64 {
	if s == "" {
		return def
	}
	v, err := strconv.ParseInt(s, 10, 64)
	if err != nil {
		return def
	}
	return v
}
```

```ts
/**
 * 趋势时间范围工具。
 *
 * 后端时序数据采用固定滚动窗口（默认 60 分钟），早于窗口起点的数据已被清理。
 * 前端可选范围必须收敛到窗口内，否则用户选了范围却拿不到数据。
 */

/** 默认保留窗口（分钟），与后端 --db-retention-minutes 默认值一致。 */
export const DEFAULT_RETENTION_MIN = 60

/** 最小可选范围（分钟）。 */
const MIN_RANGE_MIN = 5

/** 候选范围档位，实际可选项由保留窗口裁剪得出。 */
const CANDIDATE_RANGES = [5, 15, 30, 60, 120, 240]

export interface RangeOption {
  label: string
  value: number
  /** 该选项是否正好等于保留窗口上限，界面据此加提示。 */
  isWindowLimit: boolean
}

/**
 * 把请求的时间范围收敛到保留窗口内。
 * @param rangeMin 期望范围（分钟）
 * @param retentionMin 后端保留窗口（分钟），非正数视为未知并原样返回
 */
export function clampRangeToWindow(rangeMin: number, retentionMin: number): number {
  if (!Number.isFinite(rangeMin) || rangeMin <= 0) return MIN_RANGE_MIN
  if (!Number.isFinite(retentionMin) || retentionMin <= 0) return rangeMin
  return Math.min(rangeMin, retentionMin)
}

/** 格式化档位标签：60 分钟以内用 m，其余用 h。 */
function formatLabel(min: number): string {
  return min < 60 ? `${min}m` : `${min / 60}h`
}

/**
 * 生成可选范围列表，超出保留窗口的档位被剔除。
 * @param retentionMin 后端保留窗口（分钟），非正数按默认 60 处理
 */
export function buildRangeOptions(retentionMin: number): RangeOption[] {
  const limit =
    Number.isFinite(retentionMin) && retentionMin > 0 ? retentionMin : DEFAULT_RETENTION_MIN

  const usable = CANDIDATE_RANGES.filter(v => v <= limit)
  if (usable.length === 0) usable.push(MIN_RANGE_MIN)

  return usable.map(v => ({
    label: formatLabel(v),
    value: v,
    isWindowLimit: v === usable[usable.length - 1],
  }))
```

```ts
/**
 * 数据持久化相关接口。
 *
 * 设计约定：后端可能未启用数据库（--db=off），此时接口返回 503。
 * 所有函数把「未启用」表达为可判断的返回值或抛错，由调用方决定降级行为，
 * 不在此处静默兜底，避免掩盖问题。
 */
import { http } from './index'

/** 后端返回的原始状态字段。 */
interface RawDbStatus {
  enabled?: boolean
  path?: string
  size_bytes?: number
  retention_minutes?: number
  sample_interval_ms?: number
  written?: number
  dropped?: number
  instances?: number
}

/** 归一化后的数据库状态。 */
export interface DbStatus {
  enabled: boolean
  sizeBytes: number
  retentionMinutes: number
  sampleIntervalMs: number
  written: number
  dropped: number
  instances: number
}

/** 默认保留窗口，与后端默认值一致。 */
const DEFAULT_RETENTION_MIN = 60

/** 把后端响应归一化，异常响应视为未启用。 */
export function normalizeDbStatus(raw: RawDbStatus | null | undefined): DbStatus {
  return {
    enabled: Boolean(raw?.enabled),
    sizeBytes: raw?.size_bytes ?? 0,
    retentionMinutes: raw?.retention_minutes ?? DEFAULT_RETENTION_MIN,
    sampleIntervalMs: raw?.sample_interval_ms ?? 1000,
    written: raw?.written ?? 0,
    dropped: raw?.dropped ?? 0,
    instances: raw?.instances ?? 0,
  }
}

export interface HistoryParams {
  ioas: string
  from: number
  to: number
  limit: number
}

/** 构造历史查询参数，IOA 去重并保持顺序。 */
export function buildHistoryParams(
  ioas: number[],
  from: number,
  to: number,
  limit: number,
): HistoryParams {
  const uniq: number[] = []
  const seen = new Set<number>()
  for (const v of ioas) {
    if (!seen.has(v)) {
      seen.add(v)
      uniq.push(v)
    }
  }
  return { ioas: uniq.join(','), from, to, limit }
}

/** 构造跨实例查询的 points 参数。 */
export function buildCrossPointsParam(refs: { instId: string; ioa: number }[]): string {
  return refs.map(r => `${r.instId}:${r.ioa}`).join(',')
}

/** 单测点历史序列，samples 为 [时间戳, 值] 数组，直接喂给 ECharts。 */
export interface HistorySeries {
  ioa: number
  samples: [number, number][]
}

export interface HistoryResponse {
  instance_id: string
  from: number
  to: number
  /** 请求范围超出保留窗口时被收敛。 */
  clamped: boolean
  /** 数据量达到 limit 被截断。 */
  truncated: boolean
  retention_minutes: number
  series: HistorySeries[]
}

/** 查询数据库状态。请求失败时返回未启用状态，供页面安全降级。 */
export async function getDbStatus(): Promise<DbStatus> {
  try {
    const res = await http.get('/db/status')
    return normalizeDbStatus(res.data)
  } catch {
    return normalizeDbStatus(null)
  }
}

/** 查询单实例历史。数据库未启用时后端返回 503，此处向上抛出。 */
export async function getHistory(
  instanceId: string,
  ioas: number[],
  from: number,
  to: number,
  limit = 5000,
): Promise<HistoryResponse> {
  const res = await http.get(`/instances/${instanceId}/history`, {
    params: buildHistoryParams(ioas, from, to, limit),
  })
  return res.data
}

export interface LatestPoint {
  ioa: number
  ts: number
  value: number
  bool_value: boolean
  int_value: number
  qds: number
}

/** 查询实例最近快照，用于详情页首屏立即渲染。 */
export async function getLatestSnapshot(
  instanceId: string,
): Promise<{ instance_id: string; points: LatestPoint[]; source: string }> {
  const res = await http.get(`/instances/${instanceId}/snapshot/latest`)
  return res.data
}

/** 跨实例历史查询。 */
export async function getCrossHistory(
  refs: { instId: string; ioa: number }[],
  from: number,
  to: number,
  limit = 5000,
): Promise<{ from: number; to: number; series: { instance_id: string; ioa: number; samples: [number, number][] }[] }> {
  const res = await http.get('/history/cross', {
    params: { points: buildCrossPointsParam(refs), from, to, limit },
  })
  return res.data
}
```

```ts
/**
 * 数据库能力探测。
 *
 * 全局只探测一次并缓存：数据库开关在服务运行期间不会变化。
 * 页面据此决定是否展示历史回填、快照等依赖数据库的功能。
 */
import { ref, readonly } from 'vue'
import { getDbStatus, type DbStatus } from '../api/store'

const status = ref<DbStatus>({
  enabled: false,
  sizeBytes: 0,
  retentionMinutes: 60,
  sampleIntervalMs: 1000,
  written: 0,
  dropped: 0,
  instances: 0,
})

let probed = false
let inflight: Promise<void> | null = null

/** 探测一次数据库能力，并发调用共享同一请求。 */
export function useDbCapability() {
  async function probe(force = false): Promise<void> {
    if (probed && !force) return
    if (inflight && !force) return inflight

    inflight = (async () => {
      status.value = await getDbStatus()
      probed = true
      inflight = null
    })()
    return inflight
  }

  return {
    dbStatus: readonly(status),
    probe,
  }
}
```

```ts
const { dbStatus, probe: probeDb } = useDbCapability()

/**
 * 从数据库回填历史数据，再接实时增量。
 * 数据库未启用时静默跳过，退化为仅内存累积（与旧版本行为一致）。
 */
async function backfillHistory() {
  if (panelTraces.value.length === 0) return
  await probeDb()
  if (!dbStatus.value.enabled) return

  const span = clampRangeToWindow(props.timeRange, dbStatus.value.retentionMinutes)
  const to = Date.now()
  const from = to - span * 60 * 1000

  // 按实例分组，减少请求数
  const byInst = new Map<string, number[]>()
  for (const t of panelTraces.value) {
    if (!byInst.has(t.instId)) byInst.set(t.instId, [])
    byInst.get(t.instId)!.push(t.ioa)
  }

  let clamped = false
  let truncated = false

  for (const [instId, ioas] of byInst) {
    try {
      const res = await getHistory(instId, ioas, from, to)
      for (const s of res.series) {
        const tr = panelTraces.value.find(t => t.instId === instId && t.ioa === s.ioa)
        if (tr) tr.data = s.samples
      }
      if (res.clamped) clamped = true
      if (res.truncated) truncated = true
    } catch {
      // 单个实例失败不影响其他实例，继续
    }
  }

  if (truncated) {
    ElMessage.warning('历史数据量较大已截断，建议缩小时间范围')
  } else if (clamped) {
    ElMessage.info(`仅保留最近 ${dbStatus.value.retentionMinutes} 分钟数据，已按此范围展示`)
  }
  updateChart()
}
```
